create table publica-los (
  id uuid not null default gen_random_uuid (),
  state text not null,
  city text null,
  alert_type text not null,
  threshold integer null,
  title text not null,
  description text null,
  is_active boolean null default true,
  priority text null default 'medium'::text,
  created_at timestamp without time zone null default now(),
  updated_at timestamp without time zone null default now(),
  created_by text null,
  last_triggered timestamp without time zone null,
  constraint alerts_pkey primary key (id)
) TABLESPACE pg_default;

create index IF not exists idx_alerts_state on public.alerts using btree (state) TABLESPACE pg_default;

create index IF not exists idx_alerts_city on public.alerts using btree (city) TABLESPACE pg_default;

create index IF not exists idx_alerts_active on public.alerts using btree (is_active) TABLESPACE pg_default;

create index IF not exists idx_alerts_type on public.alerts using btree (alert_type) TABLESPACE pg_default;

create table public.cities (
  id serial not null,
  state_code text not null,
  name text not null,
  ibge_code text null,
  created_at timestamp without time zone null default now(),
  lat numeric(10, 8) null,
  lng numeric(11, 8) null,
  constraint cities_pkey primary key (id),
  constraint cities_state_code_name_key unique (state_code, name),
  constraint cities_state_code_fkey foreign KEY (state_code) references states (code) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_cities_state_code on public.cities using btree (state_code) TABLESPACE pg_default;

create index IF not exists idx_cities_name on public.cities using btree (name) TABLESPACE pg_default;

create table public.dengue_cases (
  id uuid not null default gen_random_uuid (),
  state text not null,
  city text not null,
  cases integer null default 0,
  deaths integer null default 0,
  confirmed_cases integer null default 0,
  suspected_cases integer null default 0,
  incidence_rate numeric(10, 2) null,
  date date not null,
  last_updated timestamp without time zone null default now(),
  created_at timestamp without time zone null default now(),
  constraint dengue_cases_pkey primary key (id),
  constraint dengue_cases_state_city_date_key unique (state, city, date)
) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_state_city on public.dengue_cases using btree (state, city) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_date on public.dengue_cases using btree (date) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_state_city_date on public.dengue_cases using btree (state, city, date desc) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_state_date on public.dengue_cases using btree (state, date desc) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_city_date on public.dengue_cases using btree (city, date desc) TABLESPACE pg_default;

create index IF not exists idx_dengue_cases_date_desc on public.dengue_cases using btree (date desc) TABLESPACE pg_default;

create table public.states (
  id serial not null,
  code text not null,
  name text not null,
  region text null,
  created_at timestamp without time zone null default now(),
  constraint states_pkey primary key (id),
  constraint states_code_key unique (code)
) TABLESPACE pg_default;

create table public.user_alerts (
  id uuid not null default gen_random_uuid (),
  user_id uuid not null,
  state text not null,
  city text null,
  alert_type text not null,
  threshold integer null,
  is_active boolean null default true,
  last_triggered timestamp without time zone null,
  created_at timestamp without time zone null default now(),
  updated_at timestamp without time zone null default now(),
  constraint user_alerts_pkey primary key (id),
  constraint user_alerts_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_user_alerts_user_id on public.user_alerts using btree (user_id) TABLESPACE pg_default;

create table public.users (
  id uuid not null default gen_random_uuid (),
  email text not null,
  full_name text null,
  address text null,
  city text null,
  state text null,
  latitude numeric(10, 8) null,
  longitude numeric(11, 8) null,
  created_at timestamp without time zone null default now(),
  updated_at timestamp without time zone null default now(),
  user_id uuid not null,
  constraint users_pkey primary key (id),
  constraint users_email_key unique (email),
  constraint users_user_id_key unique (user_id)
) TABLESPACE pg_default;
